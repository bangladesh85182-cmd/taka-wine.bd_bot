  <script>
    // ইউজারের ইউনিক আইডি ও ব্যালেন্স হ্যান্ডেল করা
    let userId = localStorage.getItem('app_user_id');
    if (!userId) {
      userId = 'USER_' + Math.floor(100000 + Math.random() * 900000);
      localStorage.setItem('app_user_id', userId);
    }
    document.getElementById('user-id-display').innerText = 'ID: ' + userId;

    let balance = parseFloat(localStorage.getItem('user_balance')) || 0.00;
    let completedTasks = JSON.parse(localStorage.getItem('completed_tasks')) || [];

    // ইন্টারফেস আপডেটের ফাংশন
    function updateUI() {
      document.getElementById('user-balance').innerText = balance.toFixed(2);
      localStorage.setItem('user_balance', balance);

      completedTasks.forEach(taskId => {
        const btn = document.getElementById(taskId + '-btn');
        if (btn) {
          btn.innerText = "সম্পন্ন";
          btn.disabled = true;
        }
      });
    }

    // ১. অ্যাড দেখার ফাংশন (৳০.৫০)
    function watchAd() {
      const btn = document.getElementById('watch-ad-btn');
      btn.disabled = true;
      let seconds = 5;

      const timer = setInterval(() => {
        btn.innerText = `বিজ্ঞাপন দেখা হচ্ছে... (${seconds}s)`;
        seconds--;

        if (seconds < 0) {
          clearInterval(timer);
          balance += 0.50;
          updateUI();
          alert("অভিনন্দন! বিজ্ঞাপন দেখার জন্য আপনি ৫০ পয়সা (৳০.৫০) পেয়েছেন।");
          btn.innerText = "বিজ্ঞাপন দেখুন (৳০.৫০)";
          btn.disabled = false;
        }
      }, 1000);
    }

    // ২. টাস্ক কমপ্লিট করার ফাংশন (৳১.০০)
    function completeTask(taskId, reward) {
      if (completedTasks.includes(taskId)) return;

      // ডেমো হিসেবে কাজ ওপেন করা
      window.open('https://google.com', '_blank');

      setTimeout(() => {
        balance += reward;
        completedTasks.push(taskId);
        localStorage.setItem('completed_tasks', JSON.stringify(completedTasks));
        updateUI();
        alert(`টাস্ক সফলভাবে শেষ হয়েছে! আপনার অ্যাকাউন্টে ৳${reward.toFixed(2)} যুক্ত করা হয়েছে।`);
      }, 2000);
    }

    // ৩. রেফারেল লিংক কপি (৳১০.০০ প্রতি রেফার)
    function copyReferralLink() {
      const currentUrl = window.location.href.split('?')[0];
      const refLink = `${currentUrl}?ref=${userId}`;
      
      navigator.clipboard.writeText(refLink);
      alert("আপনার ইউনিক রেফারেল লিংক কপি হয়েছে!\n\n" + refLink + "\n\nযেকোনো ইউজার এই লিংকে প্রবেশ করলে আপনি ৳১০.০০ পাবেন।");
    }

    // ৪. উইথড্র ফাংশন (সর্বনিম্ন ৳১০০)
    function requestWithdraw() {
      const amount = parseFloat(document.getElementById('withdraw-amount').value);
      const number = document.getElementById('account-number').value.trim();
      const method = document.getElementById('payment-method').value;

      if (!amount || amount < 100) {
        alert("সর্বনিম্ন উইথড্র পরিমাণ ১০০ টাকা (৳১০০)!");
        return;
      }

      if (amount > balance) {
        alert("আপনার অ্যাকাউন্টে পর্যাপ্ত ব্যালেন্স নেই!");
        return;
      }

      if (number.length < 11) {
        alert("অনুগ্রহ করে সঠিক ১১ ডিজিটের মোবাইল নম্বর দিন!");
        return;
      }

      // ব্যালেন্স কেটে নেওয়া
      balance -= amount;
      updateUI();

      alert(`আপনার ৳${amount} টাকার ${method} উইথড্র রিকোয়েস্ট সফলভাবে জমা হয়েছে। শীঘ্রই এটি প্রসেস করা হবে।`);

      // ইনপুট খালি করা
      document.getElementById('withdraw-amount').value = '';
      document.getElementById('account-number').value = '';
    }

    // অ্যাপ লোড হওয়ার পর UI সিঙ্ক করা
    updateUI();
  </script>
