    <p style="font-size: 13px; color: #666; margin: 5px 0 10px;">প্রতিটি সফল রেফারে পাবেন ৳১০.০০</p>
    <button onclick="copyReferralLink()">রেফারেল লিংক কপি করুন</button>
  </div>

  <!-- উইথড্র সেকশন -->
  <div class="card">
    <h3>💸 টাকা উত্তোলন (Withdraw)</h3>
    <p style="font-size: 12px; color: #d9534f; margin-bottom: 10px;">সর্বনিম্ন উইথড্র সীমা: ৳১০০.০০</p>
    
    <select id="payment-method">
      <option value="bkash">বিকাশ (bKash)</option>
      <option value="nagad">নগদ (Nagad)</option>

    </select>
    
    <input type="number" id="withdraw-amount" placeholder="পরিমাণ (যেমন: 100)" min="100">
    <input type="text" id="account-number" placeholder="মোবাইল নম্বর">
    
    <button onclick="requestWithdraw()" style="background: #28a745; margin-top: 5px;">উইথড্র রিকোয়েস্ট পাঠান</button>

  <script>
    const tg = window.Telegram.WebApp;
    tg.expand();

    // ইউজারের ডাটা লোড বা ইনিশিয়ালাইজ করা
    let balance = parseFloat(localStorage.getItem('user_balance')) || 0.00;
    let completedTasks = JSON.parse(localStorage.getItem('completed_tasks')) || [];

    // স্ক্রিনে ব্যালেন্স আপডেট
    function updateUI() {
      document.getElementById('user-balance').innerText = balance.toFixed(2);
      localStorage.setItem('user_balance', balance);

      if (tg.initDataUnsafe && tg.initDataUnsafe.user) {
        document.getElementById('user-name').innerText = tg.initDataUnsafe.user.first_name;
      }

      // পূর্বের সম্পন্ন করা টাস্ক ডিজেবল রাখা
      completedTasks.forEach(taskId => {
        const btn = document.getElementById(taskId + '-btn');
        if (btn) {
          btn.innerText = "সম্পন্ন";
          btn.disabled = true;
        }
      });
    }

    // ১. অ্যাড দেখার ফাংশন (৳০.৫০)
    function watchAd() {
      const btn = document.getElementById('watch-ad-btn');
      btn.disabled = true;
      let count = 5;
      
      const timer = setInterval(() => {
        btn.innerText = `অ্যাড চলছে... (${count}s)`;
        count--;
        if (count < 0) {
          clearInterval(timer);
          balance += 0.50;
          updateUI();
          alert("অভিনন্দন! আপনি ৳০.৫০ রিওয়ার্ড পেয়েছেন।");
          btn.innerText = "অ্যাড দেখুন (৳০.৫০)";
          btn.disabled = false;
        }
      }, 1000);
    }

    // ২. টাস্ক কমপ্লিট করার ফাংশন (৳১.০০)
    function completeTask(taskId, reward) {
      if (completedTasks.includes(taskId)) return;

      // উদাহরণের জন্য একটি ডামি লিংক চালু করা
      window.open('https://t.me', '_blank');

      setTimeout(() => {
        balance += reward;
        completedTasks.push(taskId);
        localStorage.setItem('completed_tasks', JSON.stringify(completedTasks));
        updateUI();
        alert(`টাস্ক সম্পন্ন হয়েছে! ৳${reward.toFixed(2)} যোগ করা হয়েছে।`);
      }, 2000);
    }

    // ৩. রেফারেল লিংক কপি (৳১০.০০ প্রতি রেফার)
    function copyReferralLink() {
      const userId = tg.initDataUnsafe?.user?.id || 'demo_user';
      const botUsername = 'your_bot_username'; // আপনার বটের ইউজারনেম দিন
      const refLink = `https://t.me/${botUsername}?start=ref_${userId}`;
      
      navigator.clipboard.writeText(refLink);
      alert("রেফারেল লিংক কপি হয়েছে!\n" + refLink + "\n\nএই লিংকে কেউ জয়েন করলে আপনি ৳১০.০০ পাবেন।");
    }

    // ৪. উইথড্র ফাংশন (সর্বনিম্ন ৳১০০)
    function requestWithdraw() {
      const amount = parseFloat(document.getElementById('withdraw-amount').value);
      const number = document.getElementById('account-number').value.trim();
      const method = document.getElementById('payment-method').value;

      if (!amount || amount < 100) {
        alert("সর্বনিম্ন উইথড্র পরিমাণ ৳১০০ টাকা!");
        return;
      }

      if (amount > balance) {
        alert("আপনার ওয়ালেটে পর্যাপ্ত ব্যালেন্স নেই!");
        return;
      }

      if (number.length < 11) {
        alert("সঠিক ১১ ডিজিটের মোবাইল নম্বর দিন!");
        return;
      }

      // ব্যালেন্স কেটে নেওয়া
      balance -= amount;
      updateUI();

      alert(`আপনার ৳${amount} টাকার ${method.toUpperCase()} উইথড্র রিকোয়েস্ট সফলভাবে জমা হয়েছে। শীঘ্রই বিকাশ/নগদে পাঠানো হবে।`);
      
      // ইনপুট ফিল্ড খালি করা
      document.getElementById('withdraw-amount').value = '';
      document.getElementById('account-number').value = '';
    }

    // অ্যাপ লোড হলে UI সেট করা
    updateUI();
  </script>
